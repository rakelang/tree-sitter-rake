record Point {
  i32: x, y;
}

extern slow log_value(value: i32) from "helper.h"
state turns: i32 := 0
embed weights from "student.bin"
const limit: i32 = 8

run scale(x: []f32, out: mut []f32, <n: i32>):
  for <i: i32> from <0> up to <n> by <4>:
    out[<i>] <- if x[<i>] > 0.0 then x[<i>] else -x[<i>]

slow main() -> i32:
  p: Point := Point { x: 1, y: 2 }
  values: [8]f32 := [0.0; 8]
  while p.x < limit:
    if p.x = 3:
      break
    else if p.y > 4:
      continue
    else:
      p.x <- wrap(i32, p.x + 1)
  turns <- turns + i32(values[unchecked 0])
  return p.x
